    function drawPriceChart(t) {
      const canvas = $("priceCanvas");
      const ctx = canvas.getContext("2d");
      const w = canvas.width;
      const h = canvas.height;
      ctx.clearRect(0, 0, w, h);

      const curPrice = (t.ngnReserve || 250000) / Math.max(1, t.tokenReserve || 1000000);
      let pts = Array.isArray(t.priceHistory) && t.priceHistory.length ? t.priceHistory.map((p) => Number(p.priceNgn || curPrice)) : [curPrice * 0.85, curPrice];
      if (pts.length === 1) pts = [pts[0] * 0.9, pts[0]];

      const minP = Math.min(...pts) * 0.92;
      const maxP = Math.max(...pts) * 1.08;
      const span = Math.max(1e-9, maxP - minP);

      // Subtle horizontal grid lines
      ctx.strokeStyle = "rgba(255,255,255,0.06)";
      ctx.lineWidth = 1;
      for (let g = 1; g <= 3; g++) {
        const y = (h / 4) * g;
        ctx.beginPath();
        ctx.moveTo(12, y);
        ctx.lineTo(w - 12, y);
        ctx.stroke();
      }

      // Line path
      ctx.strokeStyle = "#d4ff59";
      ctx.lineWidth = 2.5;
      ctx.beginPath();
      pts.forEach((p, idx) => {
        const x = 14 + (idx / (pts.length - 1)) * (w - 28);
        const y = h - 18 - ((p - minP) / span) * (h - 36);
        if (idx === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
      });
      ctx.stroke();

      // Latest point dot
      const lastX = 14 + (w - 28);
      const lastY = h - 18 - ((pts[pts.length - 1] - minP) / span) * (h - 36);
      ctx.fillStyle = "#d4ff59";
      ctx.beginPath();
      ctx.arc(lastX, lastY, 4, 0, Math.PI * 2);
      ctx.fill();
    }

    window.executeTrade = async () => {
      if (!currentUser || !profile) return openAuthModal();
      const t = tokens.find((x) => x.id === selectedTokenId);
      if (!t) return;
      const amt = parseFloat($("tradeAmountInput").value) || 0;
      if (amt <= 0) return toast("Enter an amount.");

      const holdings = { ...(profile.holdings || {}) };
      const avgBuyNgn = { ...(profile.avgBuyNgn || {}) };
      const held = Number(holdings[t.id] || 0);
      const history = Array.isArray(t.priceHistory) ? [...t.priceHistory] : [];

      if (tradeSide === "buy") {
        if ((profile.ngnBalance || 0) < amt) return toast("Insufficient NGN balance.");
        const fee = amt * APP_CFG.feeRate;
        const netNgn = amt - fee;
        const nextNgn = t.ngnReserve + netNgn;
        const nextTok = t.kNgn / nextNgn;
        const tokensOut = Math.max(0, t.tokenReserve - nextTok);
        const newPriceNgn = nextNgn / Math.max(1, nextTok);

        const prevCost = held * Number(avgBuyNgn[t.id] || newPriceNgn);
        const nextTotalTokens = held + tokensOut;
        avgBuyNgn[t.id] = nextTotalTokens > 0 ? (prevCost + amt) / nextTotalTokens : newPriceNgn;
        holdings[t.id] = nextTotalTokens;
        history.push({ t: Date.now(), priceNgn: newPriceNgn });

        const updTokBuy = {
          ngnReserve: Number(nextNgn.toFixed(2)),
          tokenReserve: Number(nextTok.toFixed(4)),
          volumeNgn: (t.volumeNgn || 0) + amt,
          priceHistory: history.slice(-30)
        };
        const updUsrBuy = {
          ngnBalance: Number((profile.ngnBalance - amt).toFixed(2)),
          holdings,
          avgBuyNgn
        };
        // Immediate local UI feedback before network round-trip finishes
        Object.assign(t, updTokBuy);
        Object.assign(profile, updUsrBuy);
        refreshWalletDisplays();
        refreshTradeView();
        toast(`Bought ${tokensOut.toFixed(2)} ${t.ticker}!`);

        Promise.all([
          updateDoc(doc(db, "tokens", t.id), updTokBuy),
          updateDoc(doc(db, "users", currentUser.uid), updUsrBuy),
          addDoc(collection(db, "trades"), {
            tokenId: t.id,
            side: "buy",
            ngnAmt: amt,
            tokensAmt: tokensOut,
            traderShort: profile.evmAddress ? `${profile.evmAddress.slice(0, 4)}...${profile.evmAddress.slice(-4)}` : "user",
            createdAt: Date.now()
          })
        ]).catch((err) => toast("Sync warning: " + err.message));
      } else {
        if (amt > held + 1e-6) return toast(`You only hold ${held.toFixed(2)} ${t.ticker}.`);
        const sellAmt = Math.min(amt, held);
        const avgBuy = Number((profile.avgBuyNgn || {})[t.id] || ((t.ngnReserve || 250000) / Math.max(1, t.tokenReserve || 1000000)));
        const costOfSold = sellAmt * avgBuy;

        const nextTok = t.tokenReserve + sellAmt;
        const nextNgn = t.kNgn / nextTok;
        const grossNgn = Math.max(0, t.ngnReserve - nextNgn);
        const netPayout = grossNgn * (1 - APP_CFG.feeRate);
        const newPriceNgn = nextNgn / Math.max(1, nextTok);
        const pctChange = costOfSold > 0 ? (((netPayout - costOfSold) / costOfSold) * 100) : 0;
        const pctLabel = `${pctChange >= 0 ? "+" : ""}${pctChange.toFixed(1)}%`;

        holdings[t.id] = Math.max(0, held - sellAmt);
        if (holdings[t.id] === 0) delete avgBuyNgn[t.id];
        history.push({ t: Date.now(), priceNgn: newPriceNgn });

        const updTokSell = {
          ngnReserve: Number(nextNgn.toFixed(2)),
          tokenReserve: Number(nextTok.toFixed(4)),
          volumeNgn: (t.volumeNgn || 0) + grossNgn,
          priceHistory: history.slice(-30)
        };
        const updUsrSell = {
          ngnBalance: Number(((profile.ngnBalance || 0) + netPayout).toFixed(2)),
          holdings,
          avgBuyNgn
        };
        // Immediate local UI feedback before network round-trip finishes
        Object.assign(t, updTokSell);
        Object.assign(profile, updUsrSell);
        refreshWalletDisplays();
        refreshTradeView();
        toast(`Sold ${sellAmt.toFixed(2)} ${t.ticker} (${pctLabel}) → ₦${netPayout.toFixed(2)} returned!`);

        Promise.all([
          updateDoc(doc(db, "tokens", t.id), updTokSell),
          updateDoc(doc(db, "users", currentUser.uid), updUsrSell),
          addDoc(collection(db, "trades"), {
            tokenId: t.id,
            side: "sell",
            ngnAmt: netPayout,
            tokensAmt: sellAmt,
            traderShort: profile.evmAddress ? `${profile.evmAddress.slice(0, 4)}...${profile.evmAddress.slice(-4)}` : "user",
            createdAt: Date.now()
          })
        ]).catch((err) => toast("Sync warning: " + err.message));
      }

      updateTradeQuote();
    };

    function bytesToB64(arr) {
      let s = "";
      for (let i = 0; i < arr.length; i++) s += String.fromCharCode(arr[i]);
      return btoa(s);
    }

    function b64ToBytes(b64) {
      const s = atob(b64);
      const arr = new Uint8Array(s.length);
      for (let i = 0; i < s.length; i++) arr[i] = s.charCodeAt(i);
      return arr;
    }

    async function deriveAesKey(password, saltB64) {
      const enc = new TextEncoder();
      const keyMaterial = await crypto.subtle.importKey(
        "raw", enc.encode(password), "PBKDF2", false, ["deriveKey"]
      );
      return crypto.subtle.deriveKey(
        { name: "PBKDF2", salt: b64ToBytes(saltB64), iterations: 120000, hash: "SHA-256" },
        keyMaterial,
        { name: "AES-GCM", length: 256 },
        false,
        ["encrypt", "decrypt"]
      );
    }

    async function encryptSecret(plainText, password, existingSaltB64) {
      const salt = existingSaltB64 ? b64ToBytes(existingSaltB64) : crypto.getRandomValues(new Uint8Array(16));
      const saltB64 = existingSaltB64 || bytesToB64(salt);
      const iv = crypto.getRandomValues(new Uint8Array(12));
      const aesKey = await deriveAesKey(password, saltB64);
      const cipherBuf = await crypto.subtle.encrypt(
        { name: "AES-GCM", iv }, aesKey, new TextEncoder().encode(plainText)
      );
      return {
        cipherB64: bytesToB64(new Uint8Array(cipherBuf)),
        ivB64: bytesToB64(iv),
        saltB64: saltB64
      };
    }

    async function decryptSecret(encObj, password) {
      if (!encObj || !encObj.cipherB64 || !encObj.ivB64 || !encObj.saltB64) return null;
      const aesKey = await deriveAesKey(password, encObj.saltB64);
      const plainBuf = await crypto.subtle.decrypt(
        { name: "AES-GCM", iv: b64ToBytes(encObj.ivB64) }, aesKey, b64ToBytes(encObj.cipherB64)
      );
      return new TextDecoder().decode(plainBuf);
    }

    window.unlockOwnerVault = async function() {
      if (!currentUser || !profile) return;
      const pass = $("vaultPassInput").value;
      if (!pass || pass.length < 6) return toast("Enter an owner vault password (6+ chars).");

      try {
        if (!profile.vaultPassSalt || !profile.vaultPassVerifier) {
          const salt = crypto.getRandomValues(new Uint8Array(16));
          const saltB64 = bytesToB64(salt);
          const verifier = await encryptSecret("NAIJA_LAUNCH_OWNER_VERIFIED", pass, saltB64);
          await updateDoc(doc(db, "users", currentUser.uid), {
            vaultPassSalt: saltB64,
            vaultPassVerifier: verifier
          });
          vaultSessionKey = await deriveAesKey(pass, saltB64);
          vaultSessionPass = pass;
          toast("Owner vault password set & encrypted!");
        } else {
          const check = await decryptSecret(profile.vaultPassVerifier, pass);
          if (check !== "NAIJA_LAUNCH_OWNER_VERIFIED") {
            return toast("Incorrect owner vault password.");
          }
          vaultSessionKey = await deriveAesKey(pass, profile.vaultPassSalt);
          vaultSessionPass = pass;
          toast("Owner vaults unlocked.");
        }
        $("vaultLockScreen").classList.add("hidden");
        $("vaultUnlockedScreen").classList.remove("hidden");
        renderOwnerCoinVaults();
      } catch (err) {
        toast("Incorrect password or decryption failed.");
      }
    };

    window.toggleRevealVaultKey = async function(tokenId) {
      if (!vaultSessionPass) return toast("Unlock vault first.");
      if (revealedKeyTokenIds[tokenId]) {
        delete revealedKeyTokenIds[tokenId];
        renderOwnerCoinVaults();
        return;
      }
      const t = tokens.find(function(x) { return x.id === tokenId; });
      if (!t || !t.encryptedVaultKey) return toast("No encrypted key stored for this coin.");
      try {
        const pk = await decryptSecret(t.encryptedVaultKey, vaultSessionPass);
        revealedKeyTokenIds[tokenId] = pk;
        renderOwnerCoinVaults();
      } catch (e) {
        toast("Could not decrypt key with current password.");
      }
    };

    window.refundCoinHolders = async function(tokenId) {
      if (!vaultSessionPass) return toast("Unlock owner vault first.");
      const t = tokens.find(function(x) { return x.id === tokenId; });
      if (!t) return;
      const curPrice = (t.ngnReserve || 0) / Math.max(1, t.tokenReserve || 1000000);
      let refundedTotal = 0;

      for (const u of allUsersCache) {
        const held = Number((u.holdings && u.holdings[tokenId]) || 0);
        if (held > 0) {
          const refundNgn = Number((held * curPrice).toFixed(2));
          const nextHoldings = Object.assign({}, u.holdings || {});
          delete nextHoldings[tokenId];
          const nextAvg = Object.assign({}, u.avgBuyNgn || {});
          delete nextAvg[tokenId];
          await updateDoc(doc(db, "users", u.id), {
            ngnBalance: Number(((u.ngnBalance || 0) + refundNgn).toFixed(2)),
            holdings: nextHoldings,
            avgBuyNgn: nextAvg
          });
          refundedTotal += refundNgn;
        }
      }

      await updateDoc(doc(db, "tokens", tokenId), {
        ngnReserve: Number(Math.max(0, (t.ngnReserve || 0) - refundedTotal).toFixed(2))
      });
      toast("Refunded NGN " + refundedTotal.toLocaleString() + " to " + t.ticker + " holders.");
      renderOwnerCoinVaults();
    };
